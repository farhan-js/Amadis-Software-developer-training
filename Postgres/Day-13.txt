CREATE OR REPLACE FUNCTION display_cars(x INT) 
RETURNS TABLE(
	brand VARCHAR,
	model VARCHAR
)
AS $$
BEGIN
RETURN QUERY SELECT c.brand, c.model FROM "Day 1".cars AS c WHERE year = x;
END;
$$
LANGUAGE plpgsql;


SELECT * FROM display_cars(2022);


CREATE PROCEDURE change_year()
LANGUAGE plpgsql
AS $$
BEGIN
    UPDATE cars
    SET year = 2024;
END;
$$;
