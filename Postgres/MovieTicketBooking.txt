CREATE TABLE Movies (
    movie_id SERIAL PRIMARY KEY,
    title VARCHAR(100) NOT NULL,
    genre VARCHAR(50),
    duration_minutes INT,
    release_date DATE
);

CREATE TABLE Theatres (
    theatre_id SERIAL PRIMARY KEY,
    theatre_name VARCHAR(100) NOT NULL,
    location VARCHAR(100) NOT NULL
);

CREATE TABLE Screens (
    screen_id SERIAL PRIMARY KEY,
    theatre_id INT NOT NULL,
    screen_name VARCHAR(50) NOT NULL,
    total_seats INT NOT NULL,
    FOREIGN KEY (theatre_id) REFERENCES Theatres(theatre_id)
);

CREATE TABLE Shows (
    show_id SERIAL PRIMARY KEY,
    movie_id INT NOT NULL,
    screen_id INT NOT NULL,
    show_date DATE NOT NULL,
    show_time TIME NOT NULL,
    ticket_price NUMERIC(10,2) NOT NULL,
    FOREIGN KEY (movie_id) REFERENCES Movies(movie_id),
    FOREIGN KEY (screen_id) REFERENCES Screens(screen_id)
);

CREATE TABLE Customers (
    customer_id SERIAL PRIMARY KEY,
    customer_name VARCHAR(100) NOT NULL,
    email VARCHAR(100) UNIQUE NOT NULL,
    phone VARCHAR(15)
);

CREATE TABLE Bookings (
    booking_id SERIAL PRIMARY KEY,
    show_id INT NOT NULL,
    customer_id INT NOT NULL,
    seat_number VARCHAR(10) NOT NULL,
    booking_date TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    FOREIGN KEY (show_id) REFERENCES Shows(show_id),
    FOREIGN KEY (customer_id) REFERENCES Customers(customer_id),
    UNIQUE (show_id, seat_number)
);

CREATE TABLE Payments (
    payment_id SERIAL PRIMARY KEY,
    booking_id INT NOT NULL,
    amount NUMERIC(10,2) NOT NULL,
    payment_method VARCHAR(30) NOT NULL,
    payment_status VARCHAR(20) NOT NULL,
    payment_date TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    FOREIGN KEY (booking_id) REFERENCES Bookings(booking_id)
);

SELECT 
    s.show_id,
    m.title,
    sc.screen_name,
    sc.total_seats,
    sc.total_seats - COUNT(b.booking_id) AS available_seats
FROM Shows s
JOIN Movies m ON s.movie_id = m.movie_id
JOIN Screens sc ON s.screen_id = sc.screen_id
LEFT JOIN Bookings b ON s.show_id = b.show_id
WHERE s.show_id = 13
GROUP BY s.show_id, m.title, sc.screen_name, sc.total_seats;


SELECT 
    m.title,
    SUM(p.amount) AS total_revenue
FROM Movies m
JOIN Shows s ON m.movie_id = s.movie_id
JOIN Bookings b ON s.show_id = b.show_id
JOIN Payments p ON b.booking_id = p.booking_id
WHERE p.payment_status = 'SUCCESS'
GROUP BY m.movie_id, m.title
ORDER BY total_revenue DESC
LIMIT 5;


SELECT 
    t.theatre_name,
    DATE_TRUNC('month', p.payment_date) AS month,
    SUM(p.amount) AS monthly_revenue
FROM Theatres t
JOIN Screens sc ON t.theatre_id = sc.theatre_id
JOIN Shows s ON sc.screen_id = s.screen_id
JOIN Bookings b ON s.show_id = b.show_id
JOIN Payments p ON b.booking_id = p.booking_id
WHERE p.payment_status = 'SUCCESS'
GROUP BY t.theatre_id, t.theatre_name, DATE_TRUNC('month', p.payment_date)
ORDER BY month, monthly_revenue DESC;


SELECT 
    c.customer_id,
    c.customer_name,
    COUNT(b.booking_id) AS total_tickets
FROM Customers c
JOIN Bookings b ON c.customer_id = b.customer_id
WHERE b.booking_date >= CURRENT_DATE - INTERVAL '1 year'
GROUP BY c.customer_id, c.customer_name
HAVING COUNT(b.booking_id) > 10
ORDER BY total_tickets DESC;
